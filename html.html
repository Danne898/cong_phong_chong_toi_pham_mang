// ── auth.middleware.ts ── JWT + RBAC + 2FA Validation import { Injectable, UnauthorizedException } from '@nestjs/common'; import { JwtService } from '@nestjs/jwt'; import { totp } from 'otplib'; @Injectable() export class AuthMiddleware { constructor(private jwt: JwtService) {} async use(req, res, next) { const token = this.extractToken(req); if (!token) throw new UnauthorizedException('Missing auth token'); const payload = await this.jwt.verifyAsync(token, { algorithms: ['RS256'], // RSA-256 signing issuer: 'mps-cybercrime.gov.vn' }); // 2FA verification nếu action nhạy cảm if (req.headers['x-totp-code']) { const valid = totp.verify({ token: req.headers['x-totp-code'], secret: payload.totpSecret }); if (!valid) throw new UnauthorizedException('Invalid 2FA code'); } req.user = payload; next(); } } // ── roles.guard.ts ── RBAC Guard @Injectable() export class RolesGuard { canActivate(ctx: ExecutionContext): boolean { const required = this.reflector.get('roles', ctx.getHandler()); const { user } = ctx.switchToHttp().getRequest(); return required?.includes(user.role) ?: true; } }
// ── report.service.ts ── Incident Report với SQL Injection protection @Injectable() export class ReportService { constructor( @InjectRepository(IncidentReport) private repo: Repository<IncidentReport>, private encryption: EncryptionService, private audit: AuditLogService, private notify: NotificationService ) {} async createReport(dto: CreateReportDto, user: User) { // Sanitize input chống XSS const clean = sanitizeHtml(dto.description, { allowedTags: [], allowedAttributes: {} }); // Mã hóa dữ liệu nhạy cảm trước khi lưu const report = this.repo.create({ userId: user.id, description: await this.encryption.encrypt(clean), amountLost: await this.encryption.encrypt(dto.amountLost.toString()), reportCode: generateReportCode(), ...dto }); const saved = await this.repo.save(report); // Ghi audit log không thể xóa await this.audit.log({ userId: user.id, action: 'CREATE_INCIDENT_REPORT', resourceId: saved.id }); // Tự động phân loại và notify điều tra viên await this.notify.assignToInvestigator(saved); return saved; } }
// ── rate-limit.guard.ts ── Rate Limiting theo IP + User @Injectable() export class RateLimitGuard implements CanActivate { constructor(private redis: RedisService) {} async canActivate(ctx: ExecutionContext): Promise<boolean> { const req = ctx.switchToHttp().getRequest(); const ip = req.ip; const key = `ratelimit:${ip}:${getMinuteBucket()}`; const current = await this.redis.incr(key); if (current === 1) await this.redis.expire(key, 60); // Giới hạn: 100 req/phút cho public, 500 cho authenticated const limit = req.user ? 500 : 100; if (current > limit) { // Auto-blacklist IP nếu vượt 10x ngưỡng if (current > limit * 10) await this.blacklistIp(ip); throw new TooManyRequestsException(); } return true; } private async blacklistIp(ip: string) { await this.redis.setex(`blacklist:${ip}`, 86400, '1'); // Ghi SIEM alert this.emitSecurityEvent({ type: 'IP_BLACKLISTED', ip }); } }
// ── encryption.service.ts ── AES-256-GCM Encryption import { createCipheriv, randomBytes, createDecipheriv } from 'crypto'; @Injectable() export class EncryptionService { private readonly ALGORITHM = 'aes-256-gcm'; private readonly KEY: Buffer; constructor() { // Key lấy từ HashiCorp Vault, không hardcode this.KEY = Buffer.from(process.env.ENCRYPTION_KEY, 'hex'); } encrypt(plaintext: string): string { const iv = randomBytes(12); // 96-bit IV cho GCM const cipher = createCipheriv(this.ALGORITHM, this.KEY, iv); const encrypted = Buffer.concat([ cipher.update(plaintext, 'utf8'), cipher.final() ]); const authTag = cipher.getAuthTag(); // GCM Auth Tag chống tamper // Format: iv:authTag:ciphertext (base64) return [ iv.toString('base64'), authTag.toString('base64'), encrypted.toString('base64') ].join(':'); } decrypt(ciphertext: string): string { const [ivB64, tagB64, dataB64] = ciphertext.split(':'); const decipher = createDecipheriv( this.ALGORITHM, this.KEY, Buffer.from(ivB64, 'base64') ); decipher.setAuthTag(Buffer.from(tagB64, 'base64')); return Buffer.concat([ decipher.update(Buffer.from(dataB64, 'base64')), decipher.final() ]).toString('utf8'); } }

❓ Câu Hỏi Thường Gặp

Làm sao biết mình đang bị lừa đảo qua điện thoại? ▼
Các dấu hiệu điển hình: người gọi tự xưng là công an/viện kiểm sát yêu cầu chuyển tiền vào "tài khoản bảo lãnh"; áp lực thời gian, yêu cầu giữ bí mật; yêu cầu cung cấp OTP, PIN tài khoản ngân hàng; không cho phép trao đổi với người thân. Cơ quan Nhà nước KHÔNG BAO GIỜ yêu cầu chuyển tiền qua điện thoại.
Sau khi nộp báo cáo, bao lâu được phản hồi? ▼
Báo cáo sự cố khẩn cấp được xử lý trong vòng 15 phút. Báo cáo thông thường được gán điều tra viên trong vòng 2 giờ. Bạn sẽ nhận được mã báo cáo để theo dõi tiến độ qua cổng hoặc SMS. Thời gian điều tra đầy đủ phụ thuộc vào mức độ phức tạp, thường từ 3-30 ngày.
Dữ liệu của tôi có được bảo mật không? ▼
Toàn bộ dữ liệu được mã hóa AES-256-GCM trong cơ sở dữ liệu. Thông tin cá nhân (họ tên, SĐT, email) được mã hóa ở cấp độ trường dữ liệu. Chỉ điều tra viên có thẩm quyền mới xem được thông tin đầy đủ. Mọi truy cập đều được ghi audit log không thể xóa.
Tôi có thể báo cáo ẩn danh không? ▼
Có, hệ thống hỗ trợ báo cáo ẩn danh cho một số loại sự cố. Tuy nhiên, báo cáo có thông tin liên hệ sẽ giúp điều tra viên liên hệ lấy thêm bằng chứng và tăng tỷ lệ thành công trong truy tố. Thông tin của bạn được bảo vệ theo Luật An toàn thông tin mạng.
Website lừa đảo được xử lý như thế nào? ▼
Sau khi xác minh, VNPT/Viettel và các nhà cung cấp DNS trong nước được thông báo để chặn truy cập. Đồng thời phối hợp với ICANN và hosting provider quốc tế để takedown domain/server. Thời gian trung bình để chặn 1 website lừa đảo là 4-8 giờ.
